@AndroidEntryPoint
class MainActivity : ComponentActivity() {

    private val viewModel: TransformerViewModel by viewModels()
    private val accountViewModel: AccountViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        enableEdgeToEdge()
        super.onCreate(savedInstanceState)
        setContent {
            TransformerMonitoringSystemTheme {
                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = MaterialTheme.colorScheme.background
                ) {
                    val authState by accountViewModel.authState.collectAsState()

                    when (authState) {
                        AuthUiState.Initializing -> {
                            Box(
                                modifier = Modifier.fillMaxSize(),
                                contentAlignment = Alignment.Center
                            ) {
                                CircularProgressIndicator()
                            }
                        }

                        AuthUiState.Unauthenticated -> {
                            AuthScreen(accountViewModel = accountViewModel)
                        }

                        AuthUiState.Authenticated -> {
                            val navController = rememberNavController()
                            NavHost(
                                navController = navController,
                                startDestination = NavScreen.List.route
                            ) {
                                composable(NavScreen.List.route) {
                                    TransformersScreen(
                                        viewModel = viewModel,
                                        accountViewModel = accountViewModel,
                                        onNavigateToForm = { navController.navigate(NavScreen.Form.route) },
                                        onNavigateToDetails = { id ->
                                            navController.navigate(NavScreen.Details.createRoute(id))
                                        }
                                    )
                                }
                                composable(NavScreen.Form.route) {
                                    TransformerFormScreen(
                                        navController = navController,
                                        viewModel = viewModel
                                    )
                                }
                                composable(
                                    route = NavScreen.Details.route,
                                    arguments = listOf(navArgument("transformerId") { type = NavType.IntType })
                                ) { backStackEntry ->
                                    val id = backStackEntry.arguments?.getInt("transformerId") ?: 0
                                    TransformerDetailsScreen(
                                        transformerId = id,
                                        navController = navController,
                                        viewModel = viewModel,
                                        accountViewModel = accountViewModel
                                    )
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}

/**

interface AuthApiService {
    @POST("auth/register")
    suspend fun register(@Body credentials: AuthCredentials): AuthResponse

    @POST("auth/login")
    suspend fun login(@Body credentials: AuthCredentials): AuthResponse

    @POST("auth/logout")
    suspend fun logout(@Body request: LogoutRequest)

    @POST("auth/refresh")
    suspend fun refreshTokens(@Body request: RefreshRequest): AuthResponse

    @POST("auth/role/admin")
    suspend fun changeRoleToAdmin(@Body request: ChangeRoleToAdminRequest)

    @POST("auth/role/user")
    suspend fun changeRoleToUser()
}

/**
 * Інтерфейс REST API для роботи з профілем користувача.
 */
interface UserApiService {
    @GET("users/profile")
    suspend fun getCurrentUserProfile(): UserProfile

    @PUT("users/profile")
    suspend fun updateProfile(@Body dto: UpdateUser): UserProfile

    @DELETE("users/profile")
    suspend fun deleteProfile(): UserProfile
}

    @GET("transformers/{id}")

    @POST("transformers")
    suspend fun addTransformer(@Body transformer: Transformer): Transformer

    @PUT("transformers/{id}")
    suspend fun updateTransformer(@Path("id") id: Int, @Body transformer: Transformer): Transformer

    @DELETE("transformers/{id}")
    suspend fun deleteTransformer(@Path("id") id: Int)

    @POST("transformers/{id}/statuses")

    @DELETE("transformers/statuses/{statusId}")

}

package ua.kpi.transformermonitoringsystem.data

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.firstOrNull
import kotlinx.coroutines.withContext
import ua.kpi.transformermonitoringsystem.data.model.AuthCredentials
import ua.kpi.transformermonitoringsystem.data.model.AuthResponse
import ua.kpi.transformermonitoringsystem.data.model.ChangeRoleToAdminRequest
import ua.kpi.transformermonitoringsystem.data.model.LogoutRequest
import ua.kpi.transformermonitoringsystem.data.model.TransformerStats as TransformerStatsDto
import ua.kpi.transformermonitoringsystem.data.model.UpdateUser
import ua.kpi.transformermonitoringsystem.data.model.UserProfile
import ua.kpi.transformermonitoringsystem.database.AuthDataStore
import ua.kpi.transformermonitoringsystem.database.TransformerDao
import ua.kpi.transformermonitoringsystem.database.entity.Transformer
import ua.kpi.transformermonitoringsystem.database.entity.TransformerStatus
import ua.kpi.transformermonitoringsystem.network.ErrorHandler.retryRequest
import ua.kpi.transformermonitoringsystem.util.toDto
import ua.kpi.transformermonitoringsystem.util.toEntity

interface TransformerRepository {
    //  Session & Auth State (AuthDataStore) 
    val accessTokenFlow: Flow<String?>
    val refreshTokenFlow: Flow<String?>
    val roleFlow: Flow<String>
    val loginFlow: Flow<String?>

    //  Auth (AuthApiService + AuthDataStore) 
    suspend fun register(credentials: AuthCredentials): Result<AuthResponse>
    suspend fun login(credentials: AuthCredentials): Result<AuthResponse>
    suspend fun logout(): Result<Unit>
    suspend fun changeRoleToAdmin(secretKey: String): Result<Unit>
    suspend fun changeRoleToUser(): Result<Unit>

    //  User Profile (UserApiService + AuthDataStore) 
    suspend fun getCurrentUserProfile(): Result<UserProfile>
    suspend fun updateProfile(dto: UpdateUser): Result<UserProfile>
    suspend fun deleteProfile(): Result<UserProfile>

    //  Transformers (TransformerApiService + Room) 
    suspend fun getAllTransformers(
        forceRefresh: Boolean = false,
        localOnly: Boolean = false
    ): Result<List<Transformer>>
    suspend fun getTransformerById(
        id: Int,
        localOnly: Boolean = false
    ): Result<Transformer>
    suspend fun addTransformer(transformer: Transformer): Result<Transformer>
    suspend fun insertTransformerWithData(
        transformer: Transformer,
        statuses: List<TransformerStatus> = emptyList()
    ): Result<Transformer>
    suspend fun updateTransformer(transformer: Transformer): Result<Transformer>
    suspend fun deleteTransformer(transformer: Transformer): Result<Unit>
    suspend fun deleteTransformerById(id: Int): Result<Unit>

    //  Stats & Telemetry (TransformerApiService + Room)
    suspend fun getNetworkStats(
        transformerEntities: List<Transformer>,
        localOnly: Boolean = false
    ): Result<NetworkStats>
    fun calculateNetworkStatsLocally(transformerEntities: List<Transformer>): NetworkStats
    suspend fun getAdminStats(): Result<TransformerStatsDto>
    suspend fun getStatuses(
        transformerId: Int,
        localOnly: Boolean = false
    ): Result<List<TransformerStatus>>
    suspend fun addStatus(status: TransformerStatus): Result<TransformerStatus>
    suspend fun deleteStatus(status: TransformerStatus): Result<Unit>
    suspend fun getAverageTemperature(transformerId: Int): Result<Double>
    suspend fun getAverageLoad(transformerId: Int): Result<Double>
}

class TransformerRepositoryImpl(
    private val authApi: AuthApiService,
    private val userApi: UserApiService,
    private val transformerApi: TransformerApiService,
    private val dao: TransformerDao,
    private val authDataStore: AuthDataStore
) : TransformerRepository {

    override val accessTokenFlow: Flow<String?> = authDataStore.accessTokenFlow
    override val refreshTokenFlow: Flow<String?> = authDataStore.refreshTokenFlow
    override val roleFlow: Flow<String> = authDataStore.roleFlow
    override val loginFlow: Flow<String?> = authDataStore.loginFlow

    private suspend fun isAuthenticated(): Boolean =
        !authDataStore.accessTokenFlow.firstOrNull().isNullOrBlank()

    private suspend inline fun <T> safeCall(
        crossinline block: suspend () -> T
    ): Result<T> = withContext(Dispatchers.IO) {
        try {
            Result.success(block())
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    // AuthApiService + AuthDataStore

    override suspend fun register(credentials: AuthCredentials): Result<AuthResponse> = safeCall {
        val response = authApi.register(credentials)
        authDataStore.saveAuthData(
            accessToken = response.accessToken,
            refreshToken = response.refreshToken,
            role = "USER",
            login = credentials.login
        )
        response
    }

    override suspend fun login(credentials: AuthCredentials): Result<AuthResponse> = safeCall {
        val response = authApi.login(credentials)
        authDataStore.saveAuthData(
            accessToken = response.accessToken,
            refreshToken = response.refreshToken,
            role = "USER",
            login = credentials.login
        )
        response
    }

    override suspend fun logout(): Result<Unit> = safeCall {
        try {
            val refreshToken = authDataStore.refreshTokenFlow.firstOrNull()
            if (!refreshToken.isNullOrBlank()) {
                authApi.logout(LogoutRequest(refreshToken))
            }
        } finally {
            authDataStore.clearAuthData()
        }
    }

    override suspend fun changeRoleToAdmin(secretKey: String): Result<Unit> = withContext(Dispatchers.IO) {
        if (!isAuthenticated()) {
            return@withContext Result.failure(IllegalStateException("Користувач не авторизований"))
        }
        try {
            authApi.changeRoleToAdmin(ChangeRoleToAdminRequest(secretKey))
            authDataStore.updateRole("ADMIN")
            Result.success(Unit)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    override suspend fun changeRoleToUser(): Result<Unit> = withContext(Dispatchers.IO) {
        if (!isAuthenticated()) {
            return@withContext Result.failure(IllegalStateException("Користувач не авторизований"))
        }
        try {
            authApi.changeRoleToUser()
            authDataStore.updateRole("USER")
            Result.success(Unit)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }


    // UserApiService + AuthDataStore


    override suspend fun getCurrentUserProfile(): Result<UserProfile> = withContext(Dispatchers.IO) {
        if (!isAuthenticated()) {
            return@withContext Result.failure(IllegalStateException("Користувач не авторизований"))
        }
        try {
            val profile = retryRequest(times = 2) { userApi.getCurrentUserProfile() }
            authDataStore.updateRole(profile.role.name)
            Result.success(profile)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    override suspend fun updateProfile(dto: UpdateUser): Result<UserProfile> = withContext(Dispatchers.IO) {
        if (!isAuthenticated()) {
            return@withContext Result.failure(IllegalStateException("Користувач не авторизований"))
        }
        try {
            val updated = retryRequest(times = 2) { userApi.updateProfile(dto) }
            Result.success(updated)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    override suspend fun deleteProfile(): Result<UserProfile> = withContext(Dispatchers.IO) {
        if (!isAuthenticated()) {
            return@withContext Result.failure(IllegalStateException("Користувач не авторизований"))
        }
        try {
            val deleted = userApi.deleteProfile()
            authDataStore.clearAuthData()
            Result.success(deleted)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    // TransformerApiService + Room

    override suspend fun getAllTransformers(
        forceRefresh: Boolean,
        localOnly: Boolean
    ): Result<List<Transformer>> =
        withContext(Dispatchers.IO) {
            if (localOnly) {
                val localData = dao.getAllTransformers()
                return@withContext if (localData.isNotEmpty()) {
                    Result.success(localData)
                } else {
                    Result.failure(NoSuchElementException("Локальний кеш трансформаторів порожній"))
                }
            }

            try {
                val remoteDtos = retryRequest(times = 3) { transformerApi.getTransformers() }
                val remoteEntities = remoteDtos.map { it.toEntity() }

                val localMap = dao.getAllTransformers().associateBy { it.id }
                val remoteIds = remoteEntities.map { it.id }.toSet()

                // Видаляємо лише ті вузли, яких більше немає на сервері
                localMap.values
                    .filter { it.id !in remoteIds }
                    .forEach { dao.deleteTransformer(it) }

                remoteEntities.forEach { remote ->
                    if (localMap.containsKey(remote.id)) {
                        dao.updateTransformer(remote)
                    } else {
                        dao.insertTransformer(remote)
                    }
                }

                Result.success(remoteEntities)
            } catch (e: Exception) {
                if (!forceRefresh) {
                    val localData = dao.getAllTransformers()
                    if (localData.isNotEmpty()) {
                        return@withContext Result.success(localData)
                    }
                }
                Result.failure(e)
            }
        }

    override suspend fun getTransformerById(
        id: Int,
        localOnly: Boolean
    ): Result<Transformer> =
        withContext(Dispatchers.IO) {
            if (localOnly) {
                val localEntity = dao.getAllTransformers().find { it.id == id }
                return@withContext if (localEntity != null) {
                    Result.success(localEntity)
                } else {
                    Result.failure(NoSuchElementException("Трансформатор з ID $id відсутній у кеші"))
                }
            }

            try {
                val remoteDto = retryRequest(times = 2) { transformerApi.getTransformerById(id) }
                val entity = remoteDto.toEntity()
                dao.updateTransformer(entity)
                Result.success(entity)
            } catch (e: Exception) {
                val localEntity = dao.getAllTransformers().find { it.id == id }
                if (localEntity != null) {
                    Result.success(localEntity)
                } else {
                    Result.failure(e)
                }
            }
        }

    override suspend fun addTransformer(transformer: Transformer): Result<Transformer> =
        insertTransformerWithData(transformer, emptyList())

    override suspend fun insertTransformerWithData(
        transformer: Transformer,
        statuses: List<TransformerStatus>
    ): Result<Transformer> = withContext(Dispatchers.IO) {
        try {
            val createdDto = retryRequest(times = 2) {
                transformerApi.addTransformer(transformer.toDto())
            }
            val createdEntity = createdDto.toEntity()
            dao.insertTransformer(createdEntity)

            statuses.forEach { status ->
                val createdStatusDto = retryRequest(times = 2) {
                    transformerApi.addStatus(
                        createdEntity.id,
                        status.copy(transformerId = createdEntity.id).toDto()
                    )
                }
                dao.insertStatus(createdStatusDto.toEntity())
            }
            Result.success(createdEntity)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    override suspend fun updateTransformer(transformer: Transformer): Result<Transformer> =
        withContext(Dispatchers.IO) {
            try {
                val updatedDto = retryRequest(times = 2) {
                    transformerApi.updateTransformer(transformer.id, transformer.toDto())
                }
                val updatedEntity = updatedDto.toEntity()
                dao.updateTransformer(updatedEntity)
                Result.success(updatedEntity)
            } catch (e: Exception) {
                Result.failure(e)
            }
        }

    override suspend fun deleteTransformer(transformer: Transformer): Result<Unit> =
        withContext(Dispatchers.IO) {
            try {
                retryRequest(times = 2) { transformerApi.deleteTransformer(transformer.id) }
                dao.deleteTransformer(transformer)
                Result.success(Unit)
            } catch (e: Exception) {
                Result.failure(e)
            }
        }

    override suspend fun deleteTransformerById(id: Int): Result<Unit> =
        withContext(Dispatchers.IO) {
            try {
                retryRequest(times = 2) { transformerApi.deleteTransformer(id) }
                dao.getAllTransformers().find { it.id == id }?.let { dao.deleteTransformer(it) }
                Result.success(Unit)
            } catch (e: Exception) {
                Result.failure(e)
            }
        }

    override suspend fun getNetworkStats(
        transformerEntities: List<Transformer>,
        localOnly: Boolean
    ): Result<NetworkStats> =
        withContext(Dispatchers.IO) {
            if (localOnly) {
                return@withContext if (transformerEntities.isNotEmpty()) {
                    Result.success(calculateNetworkStatsLocally(transformerEntities))
                } else {
                    Result.failure(NoSuchElementException("Локальні дані для розрахунку показників відсутні"))
                }
            }

            try {
                val dto = retryRequest(times = 2) { transformerApi.getStats() }
                Result.success(
                    NetworkStats(
                        averageLoad = dto.averageLoad,
                        averageTemperature = dto.averageTemperature,
                        criticalCount = dto.criticalIssues
                    )
                )
            } catch (e: Exception) {
                if (transformerEntities.isNotEmpty()) {
                    Result.success(calculateNetworkStatsLocally(transformerEntities))
                } else {
                    Result.failure(e)
                }
            }
        }

    override fun calculateNetworkStatsLocally(transformerEntities: List<Transformer>): NetworkStats {
        if (transformerEntities.isEmpty()) return NetworkStats(0.0, 0.0, 0)

        val avgLoad = transformerEntities.map { it.currentLoad }.average()
        val avgTemp = transformerEntities.map { it.temperature }.average()
        val criticalCount = transformerEntities.count {
            (it.temperature > 85.0) || (it.currentLoad > (it.maxLoad * 0.9))
        }

        return NetworkStats(avgLoad, avgTemp, criticalCount)
    }

    override suspend fun getAdminStats(): Result<TransformerStatsDto> = safeCall {
        retryRequest(times = 2) { transformerApi.getStats() }
    }

    override suspend fun getStatuses(
        transformerId: Int,
        localOnly: Boolean
    ): Result<List<TransformerStatus>> =
        withContext(Dispatchers.IO) {
            if (localOnly) {
                val localStatuses = dao.getStatusesForTransformer(transformerId)
                return@withContext if (localStatuses.isNotEmpty()) {
                    Result.success(localStatuses)
                } else {
                    Result.failure(NoSuchElementException("Історія показників у кеші порожня"))
                }
            }

            try {
                val remoteDtos = retryRequest(times = 2) { transformerApi.getStatuses(transformerId) }
                val remoteEntities = remoteDtos.map { it.toEntity() }
                remoteEntities.forEach { dao.insertStatus(it) }
                Result.success(remoteEntities)
            } catch (e: Exception) {
                val localStatuses = dao.getStatusesForTransformer(transformerId)
                if (localStatuses.isNotEmpty()) {
                    Result.success(localStatuses)
                } else {
                    Result.failure(e)
                }
            }
        }

    override suspend fun addStatus(status: TransformerStatus): Result<TransformerStatus> =
        withContext(Dispatchers.IO) {
            try {
                val createdDto = retryRequest(times = 2) {
                    transformerApi.addStatus(status.transformerId, status.toDto())
                }
                val createdEntity = createdDto.toEntity()
                dao.insertStatus(createdEntity)
                Result.success(createdEntity)
            } catch (e: Exception) {
                Result.failure(e)
            }
        }

    override suspend fun deleteStatus(status: TransformerStatus): Result<Unit> =
        withContext(Dispatchers.IO) {
            try {
                retryRequest(times = 2) { transformerApi.deleteStatus(status.id) }
                dao.deleteStatus(status)
                Result.success(Unit)
            } catch (e: Exception) {
                Result.failure(e)
            }
        }

    override suspend fun getAverageTemperature(transformerId: Int): Result<Double> = safeCall {
        dao.getAverageTemperature(transformerId) ?: 0.0
    }

    override suspend fun getAverageLoad(transformerId: Int): Result<Double> = safeCall {
        dao.getAverageLoad(transformerId) ?: 0.0
    }
}

    val criticalCount: Int

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey

import kotlinx.coroutines.flow.map

/**
 * Збереження JWT токенів та ролей у DataStore.
 */
private val Context.dataStore: DataStore<Preferences> by preferencesDataStore(name = "auth_prefs")

class AuthDataStore(private val context: Context) {
    companion object {
        private val ACCESS_TOKEN_KEY = stringPreferencesKey("access_token")
        private val REFRESH_TOKEN_KEY = stringPreferencesKey("refresh_token")
        private val ROLE_KEY = stringPreferencesKey("user_role")
        private val LOGIN_KEY = stringPreferencesKey("user_login")
    }

    /**
     * Потік доступу до JWT Access Token.
     */
    val accessTokenFlow: Flow<String?> = context.dataStore.data.map { preferences ->
        preferences[ACCESS_TOKEN_KEY]?.takeIf { it.isNotBlank() }
    }

    /**
     * Потік доступу до Refresh Token.
     */
    val refreshTokenFlow: Flow<String?> = context.dataStore.data.map { preferences ->
        preferences[REFRESH_TOKEN_KEY]?.takeIf { it.isNotBlank() }
    }

    /**
     * Потік доступу до ролі користувача ("ADMIN", "USER", або "GUEST").
     */
    val roleFlow: Flow<String> = context.dataStore.data.map { preferences ->
        preferences[ROLE_KEY] ?: "GUEST"
    }

    /**
     * Потік доступу до логіну користувача.
     */
    val loginFlow: Flow<String?> = context.dataStore.data.map { preferences ->
        preferences[LOGIN_KEY]
    }

    /**
     * Збереження токенів, ролі та логін.
     */
    suspend fun saveAuthData(accessToken: String, refreshToken: String, role: String, login: String) {
        context.dataStore.edit { preferences ->
            preferences[ACCESS_TOKEN_KEY] = accessToken
            preferences[REFRESH_TOKEN_KEY] = refreshToken
            preferences[ROLE_KEY] = role
            preferences[LOGIN_KEY] = login
        }
    }

    /**
     * Оновлення ролі користувача при зміні прав.
     */
    suspend fun updateRole(role: String) {
        context.dataStore.edit { preferences ->
            preferences[ROLE_KEY] = role
        }
    }

    /**
     * Очищення даних авторизації при виході з системи (Logout).
     */
    suspend fun clearAuthData() {
        context.dataStore.edit { preferences ->
            preferences.remove(ACCESS_TOKEN_KEY)
            preferences.remove(REFRESH_TOKEN_KEY)
            preferences.remove(ROLE_KEY)
            preferences.remove(LOGIN_KEY)
        }
    }
}

import ua.kpi.transformermonitoringsystem.BuildConfig
import ua.kpi.transformermonitoringsystem.data.AuthApiService

import ua.kpi.transformermonitoringsystem.network.TokenAuthenticator
import java.io.IOException
import java.time.Instant
import java.util.concurrent.TimeUnit
import javax.inject.Provider
import javax.inject.Singleton

// Мережевий провайдер
@Module
@InstallIn(SingletonComponent::class)
object NetworkModule {

    @Provides
    @Singleton
    fun provideOkHttpClient(
        authDataStore: AuthDataStore,
        authApiProvider: Provider<AuthApiService>,
        backendHostManager: BackendHostManager,
    ): OkHttpClient {
        val candidateHosts = BuildConfig.BACKEND_HOSTS.split(",")
        val logging = HttpLoggingInterceptor().apply {
            level = HttpLoggingInterceptor.Level.BODY
        }
        return OkHttpClient.Builder()
            .connectTimeout(4, TimeUnit.SECONDS)
            .readTimeout(10, TimeUnit.SECONDS)
            .addInterceptor { chain ->
                val request = chain.request()
                val url = request.url

                val hostsToTry = mutableListOf<String>()
                val currentActive = backendHostManager.activeBackendHost
                if (!hostsToTry.contains(currentActive)) {
                    hostsToTry.add(currentActive)
                }

                candidateHosts.forEach { h ->
                    if (!hostsToTry.contains(h)) hostsToTry.add(h)
                }

                var lastException: IOException? = null
                for (host in hostsToTry) {
                    try {
                        val newUrl = url.newBuilder().host(host).build()
                        val newRequest = request.newBuilder().url(newUrl).build()
                        val response = chain.proceed(newRequest)
                        backendHostManager.activeBackendHost = host
                        return@addInterceptor response
                    } catch (e: IOException) {
                        lastException = e
                    }
                }
                throw lastException ?: IOException("Не вдалося підключитися до сервера")
            }
            .addInterceptor(AuthInterceptor(authDataStore))
            .authenticator(TokenAuthenticator(authDataStore, authApiProvider))
            .addInterceptor(logging)
            .build()
    }

    @Provides
    @Singleton
    fun provideRetrofit(okHttpClient: OkHttpClient): Retrofit {
        val gson = GsonBuilder()
            .registerTypeAdapter(Instant::class.java, object : TypeAdapter<Instant>() {
                override fun write(out: JsonWriter, value: Instant?) {
                    if (value == null) {
                        out.nullValue()
                    } else {
                        out.value(value.toString())
                    }
                }

                override fun read(reader: JsonReader): Instant? {
                    if (reader.peek() == JsonToken.NULL) {
                        reader.nextNull()
                        return null
                    }
                    val str = reader.nextString()
                    return try {
                        Instant.parse(str)
                    } catch (_: Exception) {
                        null
                    }
                }
            })
            .create()

        return Retrofit.Builder()
            .baseUrl(BuildConfig.API_BASE_URL)
            .client(okHttpClient)
            .addConverterFactory(GsonConverterFactory.create(gson))
            .build()
    }
}

package ua.kpi.transformermonitoringsystem.network

import kotlinx.coroutines.flow.firstOrNull
import kotlinx.coroutines.runBlocking
import okhttp3.Interceptor
import okhttp3.Response
import ua.kpi.transformermonitoringsystem.database.AuthDataStore

/**
 * Interceptor для автоматичної додачі заголовка авторизації Authorization: Bearer <accessToken> до кожного HTTP-запиту.
 */
class AuthInterceptor(private val authDataStore: AuthDataStore) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val original = chain.request()
        val path = original.url.encodedPath

        if (path.endsWith("auth/login") || path.endsWith("auth/register") || path.endsWith("auth/refresh")) {
            return chain.proceed(original)
        }

        val token = try {
            runBlocking { authDataStore.accessTokenFlow.firstOrNull() }
        } catch (_: Exception) {
            null
        }

        val request = if (!token.isNullOrBlank()) {
            original.newBuilder()
                .header("Authorization", "Bearer $token")
                .build()
        } else {
            original
        }
        return chain.proceed(request)
    }
}

/**
 * Централізований обробник помилок мережевих запитів.
 * Підтримує перевірку інтернет-з'єднання, логування у файл error_log.txt,
 * автоматичний механізм повторних спроб для тимчасових збоїв та диференційоване
 * інформування користувача залежно від HTTP-коду помилки.
 */
object ErrorHandler {

    /**
     * Перевірка наявності активного інтернет-з'єднання на пристрої.
     */
    fun isInternetAvailable(context: Context): Boolean {
        val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as? ConnectivityManager
            ?: return false
        val network = connectivityManager.activeNetwork ?: return false
        val activeNetwork = connectivityManager.getNetworkCapabilities(network) ?: return false
        return activeNetwork.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
    }

    /**
     * Реєстрація помилок у внутрішній файл error_log.txt у сховищі пристрою.
     */
    fun logErrorToFile(context: Context, message: String) {
        try {
            val file = File(context.filesDir, "error_log.txt")
            FileOutputStream(file, true).bufferedWriter().use { writer ->
                writer.write("${System.currentTimeMillis()}: $message\n")
            }
        } catch (e: Exception) {
            Log.e("ErrorHandler", "Cannot log error to file: ${e.localizedMessage}")
        }
    }

    /**
     * Централізована обробка винятків із перевіркою мережі та HTTP-кодів.
     * Повертає зрозуміле українськомовне повідомлення для відображення користувачеві в UI.
     */
    fun handleException(context: Context, e: Throwable): String {
        val message = when {
            !isInternetAvailable(context) -> "Відсутнє з'єднання з Інтернетом. Перевірте мережу."
            e is UnknownHostException || e is ConnectException -> "Не вдалося з'єднатися з сервером. Перевірте мережу."
            e is SocketTimeoutException -> "Перевищено час очікування відповіді від сервера (Timeout)."
            e is HttpException -> {
                when (e.code()) {
                    400 -> "Некоректний запит до сервера (HTTP 400)."
                    401 -> "Неавторизований доступ (HTTP 401). Перевірте логін та токен."
                    403 -> "Доступ заборонено (HTTP 403). У вас недостатньо прав."
                    404 -> "Запитуваний ресурс або трансформатор не знайдено (HTTP 404)."
                    500 -> "Внутрішня помилка сервера (HTTP 500). Спробуйте пізніше."
                    502, 503, 504 -> "Сервер тимчасово недоступний (HTTP ${e.code()})."
                    else -> "Помилка сервера: HTTP ${e.code()}"
                }
            }
            e is IOException -> "Помилка вводу-виводу при передачі даних по мережі."
            else -> e.localizedMessage ?: "Сталася невідома помилка"
        }
        
        Log.e("ErrorHandler", "Handled exception: $message", e)
        logErrorToFile(context, message)
        return message
    }

    /**
     * Автоматичний механізм повторення запитів при тимчасових мережевих збоях.
     */
    suspend fun <T> retryRequest(times: Int = 3, delayMs: Long = 1500L, block: suspend () -> T): T {
        var currentAttempt = 0
        var lastError: Exception? = null
        while (currentAttempt < times) {
            try {
                return block()
            } catch (e: Exception) {
                lastError = e
                currentAttempt++
                val isTransientError = (e is HttpException && e.code() in listOf(500, 502, 503, 504)) || e is IOException
                if (isTransientError && currentAttempt < times) {
                    delay(delayMs.milliseconds)
                } else {
                    break
                }
            }
        }
        throw lastError ?: Exception("Unknown error during request retry")
    }
}

package ua.kpi.transformermonitoringsystem.network

import kotlinx.coroutines.flow.firstOrNull
import kotlinx.coroutines.runBlocking
import okhttp3.Authenticator
import okhttp3.Request
import okhttp3.Response
import okhttp3.Route
import ua.kpi.transformermonitoringsystem.data.AuthApiService
import ua.kpi.transformermonitoringsystem.data.model.RefreshRequest
import ua.kpi.transformermonitoringsystem.database.AuthDataStore
import javax.inject.Provider

// Автоматичне оновлення сесії
class TokenAuthenticator(
    private val authDataStore: AuthDataStore,
    private val authApiProvider: Provider<AuthApiService>
) : Authenticator {

    override fun authenticate(route: Route?, response: Response): Request? {
        val path = response.request.url.encodedPath
        if (path.endsWith("auth/refresh") || path.endsWith("auth/login") || path.endsWith("auth/register")) {
            return null
        }

        if (responseCount(response) >= 2) {
            return null
        }

        synchronized(this) {
            val currentAccessToken = runBlocking { authDataStore.accessTokenFlow.firstOrNull() }
            val requestToken = response.request.header("Authorization")?.removePrefix("Bearer ")?.trim()

            if (!currentAccessToken.isNullOrBlank() && currentAccessToken != requestToken) {
                return response.request.newBuilder()
                    .header("Authorization", "Bearer $currentAccessToken")
                    .build()
            }

            val refreshToken = runBlocking { authDataStore.refreshTokenFlow.firstOrNull() }
            if (refreshToken.isNullOrBlank()) {
                return null
            }

            return try {
                val newTokens = runBlocking {
                    authApiProvider.get().refreshTokens(RefreshRequest(refreshToken))
                }
                if (newTokens.accessToken.isBlank() || newTokens.refreshToken.isBlank()) {
                    runBlocking { authDataStore.clearAuthData() }
                    return null
                }
                runBlocking {
                    val role = authDataStore.roleFlow.firstOrNull() ?: "USER"
                    val login = authDataStore.loginFlow.firstOrNull() ?: ""
                    authDataStore.saveAuthData(
                        accessToken = newTokens.accessToken,
                        refreshToken = newTokens.refreshToken,
                        role = role,
                        login = login
                    )
                }

                response.request.newBuilder()
                    .header("Authorization", "Bearer ${newTokens.accessToken}")
                    .build()
            } catch (_: Exception) {
                runBlocking { authDataStore.clearAuthData() }
                null
            }
        }
    }

    private fun responseCount(response: Response): Int {
        var count = 1
        var prior = response.priorResponse
        while (prior != null) {
            count++
            prior = prior.priorResponse
        }
        return count
    }
}

     * Додавання нового статусу для обраного трансформатора.
     */
    fun addStatusForSelectedTransformer(voltageStr: String, tempStr: String, loadStr: String) {
        val transformerId = _selectedTransformerId.value ?: return
        viewModelScope.launch {
            _isLoading.value = true
            _errorMessage.value = null
            try {
                val v = voltageStr.toDoubleOrNull() ?: throw Exception("Введіть коректну напругу (число)")
                val t = tempStr.toDoubleOrNull() ?: throw Exception("Введіть коректну температуру (число)")
                val l = loadStr.toDoubleOrNull() ?: throw Exception("Введіть коректне навантаження (число)")

                if (v <= 0) throw Exception("Напруга має бути більшою за 0 В")
                if (t !in -50.0..150.0) throw Exception("Температура має бути в межах від -50 до 150 °C")
                if (l < 0.0) throw Exception("Навантаження не може бути від'ємним")

                val status = TransformerStatus(voltage = v, temperature = t, load = l, transformerId = transformerId)
                repository.addStatus(status)
                    .onSuccess {
                        repository.getTransformerById(transformerId).onSuccess { updated ->
                            _selectedTransformer.value = updated
                        }
                        loadStatuses(transformerId)
                        loadData(forceRefresh = true)
                        _isSubmitted.value = true
                    }
                    .onFailure { throw it }
            } catch (e: Exception) {
                _errorMessage.value = ErrorHandler.handleException(appContext, e)
            } finally {
                _isLoading.value = false
            }
        }
    }

    /**
     * Видалення показника телеметрії.
     */
    fun deleteStatus(status: TransformerStatus) {
        viewModelScope.launch {
            repository.deleteStatus(status)
                .onSuccess {
                    val id = _selectedTransformerId.value ?: return@onSuccess
                    loadStatuses(id)
                }
                .onFailure { e ->
                    _errorMessage.value = ErrorHandler.handleException(appContext, e)
                }
        }
    }

    /**
     * Оновлення обраного трансформатора з підтримкою maxLoad.
     */
    fun updateSelectedTransformer(name: String, location: String, maxLoadStr: String) {
        val transformer = _selectedTransformer.value ?: return
        viewModelScope.launch {
            _isLoading.value = true
            _errorMessage.value = null
            _isSubmitted.value = false
            try {
                if (name.trim().length < 3) throw Exception("Назва має містити мінімум 3 символи")
                if (location.isBlank()) throw Exception("Локація не може бути порожньою")
                val maxL = maxLoadStr.toDoubleOrNull() ?: throw Exception("Введіть коректне макс. навантаження")
                if (maxL <= 0.0) throw Exception("Максимальне навантаження має бути більшим за 0")

                val updatedTransformer = transformer.copy(
                    name = name.trim(),
                    location = location.trim(),
                    maxLoad = maxL
                )
                repository.updateTransformer(updatedTransformer)
                    .onSuccess { saved ->
                        _selectedTransformer.value = saved
                        _isSubmitted.value = true
                        loadData(forceRefresh = true)
                    }
                    .onFailure { throw it }
            } catch (e: Exception) {
                _errorMessage.value = ErrorHandler.handleException(appContext, e)
            } finally {
                _isLoading.value = false
            }
        }
    }

    /**
     * Видалення обраного трансформатора.
     */
    fun deleteSelectedTransformer() {
        val transformer = _selectedTransformer.value ?: return
        viewModelScope.launch {
            _isLoading.value = true
            _errorMessage.value = null
            try {
                repository.deleteTransformer(transformer)
                loadData(forceRefresh = true)
                _isSubmitted.value = true
            } catch (e: Exception) {
                _errorMessage.value = ErrorHandler.handleException(appContext, e)
            } finally {
                _isLoading.value = false
            }
        }
    }

    fun resetSubmitState() { _isSubmitted.value = false }
    fun clearError() { _errorMessage.value = null }
}

import androidx.compose.runtime.Composable
import androidx.compose.ui.platform.LocalContext

private val DarkColorScheme = darkColorScheme(
    primary = Purple80,
    secondary = PurpleGrey80,
    tertiary = Pink80
)

private val LightColorScheme = lightColorScheme(
    primary = Purple40,
    secondary = PurpleGrey40,
    tertiary = Pink40

    /* Other default colors to override
    background = Color(0xFFFFFBFE),
    surface = Color(0xFFFFFBFE),
    onPrimary = Color.White,
    onSecondary = Color.White,
    onTertiary = Color.White,
    onBackground = Color(0xFF1C1B1F),
    onSurface = Color(0xFF1C1B1F),
    */
)

@Composable
fun TransformerMonitoringSystemTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    // Dynamic color is available on Android 12+
    dynamicColor: Boolean = true,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }

        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}
